// DETECTION RULE: VBScript stager in Temp with immediate self-delete
// Rule name: VBScript_Stager_SelfDelete_AntiForensics
// Severity: High | MITRE: T1059.005 (VBScript), T1070.004 (File Deletion)

DeviceProcessEvents
| where Timestamp > ago(1d)
| where FileName in~ ("cscript.exe","wscript.exe")
// VBS running from Temp directories
| where ProcessCommandLine has_any (
    @"\Temp\", @"\AppData\Local\Temp\", @"\Windows\Temp\"
)
| where ProcessCommandLine endswith ".vbs"
    or ProcessCommandLine has ".vbs"
// AND a del /f follows within 10 seconds (self-delete pattern)
| join kind=inner (
    DeviceProcessEvents
    | where Timestamp > ago(1d)
    | where FileName =~ "cmd.exe"
    | where ProcessCommandLine has "del" and ProcessCommandLine has "/f"
    | where ProcessCommandLine has_any (".vbs",".ps1",".bat")
    | project
        DelTime     = Timestamp,
        DelDevice   = DeviceName,
        DelCmdLine  = ProcessCommandLine
) on $left.DeviceName == $right.DelDevice
| where DelTime between (Timestamp .. (Timestamp + 10s))
| project
    Timestamp,
    DeviceName,
    AccountName,
    ScriptRun   = ProcessCommandLine,
    SelfDelete  = DelCmdLine,
    TimeDelta   = DelTime - Timestamp
| sort by Timestamp desc
